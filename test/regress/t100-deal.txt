# Verified in game 2026-10-22. Deal screen on medieval-t100 (Rome, civs met), real keys only.
# covers: **/SRA_DealView.lua **/DealSpeech.lua **/DiplomacyActionView.lua
# V lists the civs, Enter opens the overview of Dido; "Make Deal" is the 5th entry (4 x Down) while the relationship is Friendly.
# Control+Tab = previous section (Shift+Tab opens the Steam overlay).
# no-share-load: opens a diplomacy session with an AI civ
load medieval-t100
clear-advisors
action KeyNavDiplomacy -Settle 2500
expect ^Diplomacy, met civilizations
key Enter -Settle 2500
expect ^Diplomacy with
key Down -Settle 500
key Down -Settle 500
key Down -Settle 500
key Down -Settle 500
expect ^Make Deal
key Enter -Settle 3500
expect ^Deal with .* of .*Tab switches sections: Deal, Your items, Their items, Actions.*Focus: Deal, empty.
key Tab -Settle 1000
expect ^Your items, \d+ items?
key Down -Settle 800
expect ^Gold, \d+ available, 2 of \d+
key Enter -Settle 2000
expect ^Added: You give: \d+ gold\. They give nothing\.
# the player stays in the section after the AI answer (focus moves only after an action button); Control+Tab goes back to Deal
key Control+Tab -Settle 800
expect ^Deal, 1 item
key S -Settle 800
expect You give: \d+ gold
key Delete -Settle 2000
expect ^Removed: You give: \d+ gold
key Escape -Settle 2500
expect ^Deal closed
key Escape -Settle 1500
no-lua-errors
