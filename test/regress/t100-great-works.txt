# UNVERIFIED (written offline, never run in game; record with --record after the first green run). modern-start (turn 158, Rome):
# the fixtures carry no great-work facts, so check with `lua InGame` first that the player has a building with great work slots;
# if there are none, the script still checks the "no building" line. "Toggle Great Works Screen" is unbound in the user's keymap,
# so the script opens the screen like the launch bar does (LuaEvents.LaunchBar_OpenGreatWorksOverview). Tab / Control+Tab (Shift+Tab
# is the Steam overlay) jump between buildings, Space = details, V = describe the work, Escape closes. It does NOT pick up or move works.
# Needs the build with GreatWorksOverview.lua loaded ("ScreenReaderAccess: GreatWorksOverview copy loaded" in Lua.log).
# no-share-load
# covers: **/GreatWorksOverview.lua **/GreatWorkShowcase.lua
load modern-start
clear-advisors
lua InGame LuaEvents.LaunchBar_OpenGreatWorksOverview()
wait Great Works: -Timeout 10
expect ^Great Works: \d+ works?, \d+ empty slots?(, tourism plus \d+)?\. (.+: .+|You have no building that can hold great works yet), 1 of \d+
key Down -Settle 600
expect , \d+ of \d+
key Space -Settle 600
expect .{8,}
key V -Settle 600
expect .{8,}
key Tab -Settle 800
expect , \d+ of \d+
key Control+Tab -Settle 800
expect , \d+ of \d+
key End -Settle 600
expect , (\d+) of \1$
key Home -Settle 600
expect , 1 of \d+
key Escape -Settle 1500
expect ^Great Works closed
lua InGame LuaEvents.LaunchBar_OpenGreatWorksOverview()
wait Great Works: -Timeout 10
key Escape -Settle 1500
expect ^Great Works closed
no-lua-errors
