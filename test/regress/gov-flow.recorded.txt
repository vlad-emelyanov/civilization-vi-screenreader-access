# Merged scripts (one load instead of four), all on fixture bug-gov-confirm-down (Rome, no government yet; F7 opens on the Governments tab):
#  - bug-gov-confirm-down: a key sent right after Enter on a government belongs to the confirm popup, never to the list underneath.
#  - bug-gov-slots-after-change: after the change the policies page names the slots of the NEW government.
#  - gov-policy-slot-hint: Enter on an empty slot of the "My government" tab points to the Policies tab.
#  - gov-unconfirmed-escape (UNVERIFIED, blind-written, the verifier records it): unconfirmed policies were dropped silently; now the first Escape
#    warns and names the empty rows, the second closes and says the changes were discarded; Tab in between does not re-arm.
# Order: change the government once, then check the slots page, the My government hint (reached by Control+Tab from Policies), Tab back
# to Policies (the hint says "Tab to switch"), and the destructive Escape-discard part last (it closes the screen).
# Needs for the last part: >= 2 slots on the policies page and an unlocked policy for the first slot (else "No policy available for ...").
# covers: **/GovernmentScreen.lua **/InGamePopup.lua
# no-share-load: changes the government
load bug-gov-confirm-down
clear-advisors
action ToggleGovernment -Settle 2500
expect ^Governments, current: no government
#   -> PASS; window had 19 SR> line(s):
#      | Main menu
#      | Single Player
#      | Loading, please wait...
#      | ROMAN EMPIRE
#      | TRAJAN
#      | The steady hum of machinery, the acrid smell of smoke, vision clouded by ash and soot – these are the signs of changing times. The lure of scientific and cultural advancement is the engine driving your realm forward.� Now your challenge is to maintain the delicate balance between earth and man, between peace and war.
#      | Cast your net wide, oh Trajan, emperor of mighty Rome. Your legions stand at the ready to march out and establish the largest empire the world has ever seen. If you can truly get all roads to lead to Rome, yours will be an empire of great riches and luxuries.� Surely then our citizens will proclaim you as their best ruler, the Optimus Princeps.
#      | TRAJAN'S COLUMN: All cities start with an additional City Center building. (Starts with a Monument building in the Ancient era).
#      | ALL ROADS LEAD TO ROME: All cities you found or conquer start with a Trading Post. If in Trade Route range of your Capital, they also start with a road to it. Your Trade Routes earn +1 Gold for passing through Trading Posts in your own cities.
#      | LEGION: A unique land unit
#      | BATH: A unique district
#      | CONTINUE GAME. Press Enter to start.

#      | Next, 1 of 3
#      | Turn 158, 1888 AD

#      | Cannot settle here: too close to Rome. Best: 3 tiles west, 2 turns, ring 1: crabs, whales, coast, but little land, coastal flooding, recommended. X to browse.
#      | Governments, current: no government, 13 options
#      | Chiefdom, slots: 1 military, 1 economic, Enter to switch to it, No Bonus., No Bonus., 1 of 13
key Down -Settle 800
expect ^Autocracy
#   -> PASS; window had 1 SR> line(s):
#      | Autocracy, slots: 1 military, 1 economic, 1 diplomatic, 1 wildcard, Enter to switch to it, +1 to all yields for each government building and palace in a city., +10 percent Production toward Wonders., 2 of 13
key Enter -Settle 1000 -Quiet 300
expect-not (?i)^Governments, current
#   -> PASS; window had 2 SR> line(s):
#      | Popup: ARE YOU SURE?. Your people are enthusiastic to try this new form of government. Confirm the switch?
#      | No, selected. Up or Down for Yes
key Down -Settle 800
expect ^Yes
#   -> PASS; window had 1 SR> line(s):
#      | Yes
key Enter -Settle 2500 -Quiet 600
expect-not (?i)Popup: ARE YOU SURE
#   -> PASS; window had 3 SR> line(s):
#      | Government changed to Autocracy
#      | Policies, Autocracy, 4 slots, 4 empty, 5 options
#      | Military Policies slot 1: empty, 1 of 5
# --- part from bug-gov-slots-after-change ---
expect (?i)^Government changed to Autocracy
#   -> PASS; window had 3 SR> line(s):
#      | Government changed to Autocracy
#      | Policies, Autocracy, 4 slots, 4 empty, 5 options
#      | Military Policies slot 1: empty, 1 of 5
expect ^Policies, Autocracy, 4 slots, 4 empty
#   -> PASS; window had 3 SR> line(s):
#      | Government changed to Autocracy
#      | Policies, Autocracy, 4 slots, 4 empty, 5 options
#      | Military Policies slot 1: empty, 1 of 5
expect-not (?i), 0 slots
#   -> PASS; window had 3 SR> line(s):
#      | Government changed to Autocracy
#      | Policies, Autocracy, 4 slots, 4 empty, 5 options
#      | Military Policies slot 1: empty, 1 of 5
# --- part from gov-policy-slot-hint ---
clear-advisors
key Control+Tab -Settle 1200
expect ^My government, Autocracy
#   -> PASS; window had 3 SR> line(s):
#      | Notification: Fill Policy Slot
#      | My government, Autocracy, 4 slots, 4 empty, 4 options
#      | Military Policies slot 1: empty, 1 of 4
key Down -Settle 800
key Enter -Settle 1000
expect ^Policies are changed on the Policies tab\. Tab to switch\.
#   -> PASS; window had 2 SR> line(s):
#      | Economic Policies slot 1: empty, 2 of 4
#      | Policies are changed on the Policies tab. Tab to switch.
# --- part from gov-unconfirmed-escape (back on the Policies tab via Tab) ---
key Tab -Settle 1200
expect (?i)^Policies
#   -> PASS; window had 2 SR> line(s):
#      | Policies, Autocracy, 4 slots, 4 empty, 5 options
#      | Military Policies slot 1: empty, 1 of 5
key Enter -Settle 800
expect (?i)choose a policy
#   -> PASS; window had 2 SR> line(s):
#      | Military Policies slot 1, choose a policy, 18 options
#      | Chivalry: +50 percent Production toward Industrial era and earlier heavy and light cavalry units., 1 of 18
key Enter -Settle 1000
expect (?i)\d+ empty
#   -> PASS; window had 2 SR> line(s):
#      | Chivalry in Military Policies slot 1, 3 empty
#      | Military Policies slot 1: Chivalry, +50 percent Production toward Industrial era and earlier heavy and light cavalry units., 1 of 5
key Escape -Settle 1000
expect ^Policy changes not confirmed: fill the empty .* first, then confirm\. Escape again to discard them
#   -> PASS; window had 1 SR> line(s):
#      | Policy changes not confirmed: fill the empty Economic Policies and Diplomatic Policies and Wildcard Policies slots first, then confirm. Escape again to discard them.
expect-not (?i)^Government closed
#   -> PASS; window had 1 SR> line(s):
#      | Policy changes not confirmed: fill the empty Economic Policies and Diplomatic Policies and Wildcard Policies slots first, then confirm. Escape again to discard them.
key Tab -Settle 1200
key Escape -Settle 1000
expect ^Policy changes not confirmed
#   -> PASS; window had 3 SR> line(s):
#      | Governments, current: Autocracy, can't change now, 13 options
#      | Chiefdom, slots: 1 military, 1 economic, No Bonus., No Bonus., 1 of 13
#      | Policy changes not confirmed: fill the empty Economic Policies and Diplomatic Policies and Wildcard Policies slots first, then confirm. Escape again to discard them.
key Escape -Settle 1500
expect ^Government closed, policy changes discarded
#   -> PASS; window had 1 SR> line(s):
#      | Government closed, policy changes discarded
no-lua-errors
