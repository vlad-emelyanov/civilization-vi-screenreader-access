# Written blind (not yet verified in game; run with --record first). Fixture pt-modern-t161 (Rome, Oligarchy).
# A key sent right after Enter on a government belongs to the confirm popup, never to the list underneath.
# covers: **/GovernmentScreen.lua **/InGamePopup.lua
# no-share-load: opens a government change dialog
load pt-modern-t161
clear-advisors
action ToggleGovernment -Settle 2500
expect ^My government, Oligarchy
key Tab -Settle 1200
key Tab -Settle 1200
expect ^Governments, current: Oligarchy
key Enter -Settle 600 -Quiet 100
expect-not (?i)^Governments, current
# whatever Enter did (a dialog, or "can't change now"), the next Down must not move the government list
key Down -Settle 600
expect-not (?i)\d+ of 13
key Escape -Settle 1200
no-lua-errors
