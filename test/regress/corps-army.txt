# Written 2026-10-24. Corps and armies: scenario corps-army puts three Musketmen on mutually adjacent owned plots (63,35 / 64,36 / 63,36 on air-ready) and grants Nationalism + Mobilization.
# Tab menu offers Form Corps (Shift+F); X lists the neighbour; Enter forms it. The third unit then sees Form Army in its menu instead of Form Corps.
# covers: **/HubTargets.lua **/HubCursor.lua **/HubMenus.lua **/UnitPanel.lua
# no-share-load: scenario spawns units and forms a corps and an army
load air-ready
clear-advisors
# the fixture opens with the Historic moment popup (Taking Flight)
key Enter -Settle 1500
scenario corps-army
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_MUSKETMAN" and u:GetX()==63 and u:GetY()==35 then UI.SelectUnit(u) end end
key Tab -Settle 1500
expect Musketman actions
key Down -Settle 600
key Down -Settle 600
key Down -Settle 600
key Down -Settle 600
expect (?i)Form Corps, key Shift\+F
key Enter -Settle 1500
expect ^Form corps: choose a tile
key X -Settle 1500
expect valid target, unit to join. (\d+ tiles? [a-z-]+, )?Musketman
key Enter -Settle 3000
expect ^Forming a corps with Musketman
lua InGame local n=0 for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_MUSKETMAN" and u:GetMilitaryFormation()==1 then n=n+1 end end return 'corps='..n
expect-lua ^corps=1$
# the Historic moment popup (first civic of the era) swallows keys
key Enter -Settle 1500
lua InGameAccess UI.DeselectAllUnits() for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_MUSKETMAN" and u:GetMilitaryFormation()==1 then UI.SelectUnit(u) end end
key Tab -Settle 1500
expect Musketman corps, \d of \d moves
expect Musketman corps actions
# blind edit: cursor-to-selection names the formation (was plain "Musketman.")
key Escape -Settle 600
action KeyNavCursorToSelection -Settle 1500
expect Musketman corps
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_MUSKETMAN" and u:GetMilitaryFormation()==0 then UI.SelectUnit(u) end end
key Tab -Settle 1500
expect Musketman actions
key Down -Settle 600
key Down -Settle 600
key Down -Settle 600
key Down -Settle 600
expect (?i)Form Army, key Control\+Shift\+F
key Enter -Settle 1500
expect ^Form army: choose a tile
key X -Settle 1500
expect valid target, unit to join. (\d+ tiles? [a-z-]+, )?Musketman
key Enter -Settle 3000
expect ^Forming an army with Musketman
lua InGame local n=0 for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_MUSKETMAN" and u:GetMilitaryFormation()==2 then n=n+1 end end return 'army='..n
expect-lua ^army=1$
lua InGameAccess UI.DeselectAllUnits() for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_MUSKETMAN" and u:GetMilitaryFormation()==2 then UI.SelectUnit(u) end end
key Tab -Settle 1500
expect Musketman army, \d of \d moves
expect Musketman army actions
no-lua-errors
