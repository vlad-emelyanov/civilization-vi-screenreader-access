# Verified in game 2026-10-22. Scenario trader-in-capital on medieval-t100: a Trader in the capital with moves restored.
# covers: **/TradeRouteChooser.lua **/TradeOriginChooser.lua **/TradeText.lua
# Selecting a trader with moves makes the game open its TradeRouteChooser by itself (interface mode MAKE_TRADE_ROUTE); the
# Tab menu is not needed (Tab is swallowed by the chooser while it is open).
# no-share-load: scenario trader-in-capital, starts a route
load medieval-t100
scenario trader-in-capital
clear-advisors
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_TRADER" then UI.SelectUnit(u) break end end
wait Trade route from -Timeout 15
expect ^Trade route from .+: (\d+ destinations?\.|no destinations)
key Space -Settle 1000
expect .{10,}
key Enter -Settle 1000
expect ^Start trade route to .+\? Enter to confirm, Escape to go back
key Escape -Settle 1000
expect ^Back to the list
key Escape -Settle 1000
expect ^Trade route cancelled
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_TRADER" then UI.DeselectAllUnits() UI.SelectUnit(u) break end end
wait Trade route from -Timeout 15
key Enter -Settle 1000
key Enter -Settle 3000
expect ^Trade route to .+ started
no-lua-errors
