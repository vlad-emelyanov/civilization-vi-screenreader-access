# Verified 2026-10-23 on medieval-t100.
# F8 (KeyNavReports) toggles the Reports screen like the top panel button. Tab / Control+Tab (Shift+Tab is the Steam overlay) switch tabs,
# Space = details, Escape and F8 close.
# no-share-load
load medieval-t100
clear-advisors
key F8 -Settle 2500
expect ^Reports\. Yields, 1 of 4 tabs\. City income: .+: .+, 1 of \d+
key Down -Settle 600
expect , 2 of \d+
key Space -Settle 600
expect .{8,}
key End -Settle 600
expect , (\d+) of \1$
key Tab -Settle 1000
expect ^Resources, 2 of 4 tabs\. .+, 1 of \d+
key Down -Settle 600
key Space -Settle 600
expect .{8,}
key Tab -Settle 1000
expect ^City Status, 3 of 4 tabs\. .*population \d+, housing \d+.*, 1 of \d+
key Space -Settle 600
key Tab -Settle 1000
expect ^Gossip, 4 of 4 tabs\. (Turn \d+: .+|No gossip yet), 1 of \d+
key Tab -Settle 1000
expect ^Yields, 1 of 4 tabs\.
key Control+Tab -Settle 1000
expect ^Gossip, 4 of 4 tabs\.
key Escape -Settle 1500
expect ^Reports closed
key F8 -Settle 2500
expect ^Reports\. 
key F8 -Settle 1500
expect ^Reports closed
no-lua-errors
