# no-share-load: scenario spawns units, rebases and attacks
load pt-late-t221
clear-advisors
scenario air-units
# the Fighter in Aquileia (62,34): unit menu, then Re-Base to Lugdunum
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_FIGHTER" and u:GetX()==62 then UI.SelectUnit(u) end end
key Tab -Settle 1500
expect Fighter actions
key Down -Settle 600
key Down -Settle 600
key Down -Settle 600
expect Re-Base, key Shift\+R
key Escape -Settle 800
key Shift+R -Settle 1500
expect ^Rebase: choose a tile
key X -Settle 1500
expect valid target. Mediolanum, city
key X -Settle 1500
expect valid target. Lugdunum, city
key Enter -Settle 3000
expect ^Rebasing
expect Historic moment: Taking Flight
key Enter -Settle 1500
lua InGame for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_FIGHTER" and u:GetMovesRemaining()==0 then return u:GetX()..","..u:GetY() end end
expect-lua ^55,32$
# a Fighter in Lugdunum with moves left: Air Attack on the Ottoman Modern Armor 3 tiles away
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_FIGHTER" and u:GetMovesRemaining()>0 then UI.SelectUnit(u) end end
action RangedAttack -Settle 1500
expect ^Air attack: choose a tile
expect valid target, .*deals about \d+ damage.*Ottomans Modern Armor
key Enter -Settle 6000
expect ^Attacking Ottomans Modern Armor
wait Modern Armor takes \d+ damage -Timeout 20
# the Bomber
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BOMBER" then UI.SelectUnit(u) end end
action RangedAttack -Settle 1500
expect ^Air attack: choose a tile
expect valid target
key Enter -Settle 6000
expect ^Attacking
# Deploy mode (the game offers no deploy target here): the mode line and the refusal are spoken
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_FIGHTER" and u:GetMovesRemaining()>0 then UI.SelectUnit(u); UI.SetInterfaceMode(InterfaceModeTypes.DEPLOY) end end
wait Deploy: choose a tile -Timeout 8
key Enter -Settle 1500
expect Not possible there
