# fixture bug-prophet-u-cannot-activate (Great Prophet on Rome's Holy Site).
# U on a Great Prophet standing on a Holy Site runs Found Religion (same as the Tab item): the prophet is consumed, no screen opens, the Choose Religion notification says how to get to it (Control+N, Enter).
# covers: **/UnitPanel.lua **/HubMenus.lua **/HelpText.lua
# no-share-load: selects the prophet
load bug-prophet-u-cannot-activate
clear-advisors
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do local r=GameInfo.Units[u:GetUnitType()]; if r and r.UnitType=="UNIT_GREAT_PROPHET" then UI.SelectUnit(u) break end end
action ExtraHotkeysUpgrade -Settle 3000
expect-not (?i)cannot (found|activate)
wait Religion founded: choose a name and beliefs -Timeout 25
expect Religion founded: choose a name and beliefs. Control\+N
action KeyNavNotifications -Settle 1500
key Enter -Settle 2500
expect Found a religion: choose a religion
key Escape -Settle 1000
# the next selected unit is the Pikeman: U upgrades it and says only that (no "Pikeman activated")
action ExtraHotkeysUpgrade -Settle 2500
expect Upgraded to Pike and Shot
expect-not (?i)activated
no-lua-errors
