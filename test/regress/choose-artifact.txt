# Verified 2026-10-31 in the game. The ChooseArtifact popup was mouse-only. REAL path: the Archaeologist extracts via the unit actions (Excavate Artifact),
# then the popup is opened with the game's own LuaEvents.NotificationPanel_OpenArtifactPanel (the Artifact Extracted notification itself opens the Historic Moment popup instead,
# not this one). Enter picks and speaks "Artifact from X chosen"; no great-work showcase followed in this fixture (the engine had no pending choice behind the event).
# covers: **/ChooseArtifact.lua
# no-share-load: excavates with the archaeologist and opens a popup context by hand
load excavate-ready
clear-advisors
lua InGameAccess local u=Players[0]:GetUnits(); for i,x in u:Members() do if x:GetX()==49 and x:GetY()==27 and x:GetUnitType()==GameInfo.Units["UNIT_ARCHAEOLOGIST"].Index then UI.SelectUnit(x) end end
key Tab -Settle 1500
expect Jovilius actions
key Down -Settle 600
expect Excavate Artifact
key Enter -Settle 3000 -Quiet 1000
expect Excavate Artifact
lua InGame LuaEvents.NotificationPanel_OpenArtifactPanel()
# the opening lines arrive after the lua step: Space (repeats the explanation) collects them as (earlier) lines
key Space -Settle 1500
expect ^Choose artifact
expect , 1 of 2$
# Right and Up/Down move to the other civilization, announced at once
key Right -Settle 600
expect , 2 of 2$
key Down -Settle 600
expect , 1 of 2$
# Space repeats the explanation
key Space -Settle 800
expect (?i)battlefield|artifact
# Escape does not close: it says that a choice is needed
key Escape -Settle 800
expect (?i)Choose an artifact first
lua ChooseArtifact return ContextPtr:IsHidden()
expect-lua ^false$
# a map key must not reach the hub behind the popup
key J -Settle 800
lua ChooseArtifact return ContextPtr:IsHidden()
expect-lua ^false$
key Right -Settle 600
key Enter -Settle 3000 -Quiet 1000
expect ^Artifact from .+ chosen
lua ChooseArtifact return ContextPtr:IsHidden()
expect-lua ^true$
no-lua-errors
