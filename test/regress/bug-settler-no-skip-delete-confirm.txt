# Verified 2026-10-23 (needed the InGamePopup override: the game's Enter always confirmed Yes). Item 6: the Delete Unit confirmation starts on No.
# covers: **/InGamePopup.lua **/PopupDialog.lua **/UnitPanel.lua
# no-share-load: deletes/skips the unit
load bug-settler-no-skip
clear-advisors
key Enter -Settle 1500
action DeleteUnit -Settle 1500
expect (?i)^Popup: .*sure
expect (?i)^No\b
# Down = Yes, Up = No again; Enter on No keeps the unit
key Down -Settle 800
expect ^Yes
key Up -Settle 800
expect ^No
key Enter -Settle 1500
expect-not (?i)deleted
# the settler is still selected: Space says it cannot skip
action SkipTurn -Settle 1000
expect (?i)skip turn
no-lua-errors
