# Fixture bug-bomber-actions-stuck: a Bomber's Tab menu has one WMD Strike item per warhead type; they shared a menu key, so Down
# stopped at item 4 and repeated it. Now each is named by its warhead ("Nuclear Device strike") and Down reaches items 5+.
# covers: **/UnitPanel.lua **/HubMenus.lua
# no-share-load: selects the bomber, Tab menu
load bug-bomber-actions-stuck
clear-advisors
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BOMBER" then UI.SelectUnit(u) end end
key Tab -Settle 1500
expect Bomber actions
key Down -Count 2 -Settle 600
expect (?i)^nuclear device strike.*unavailable
key Down -Settle 600
expect (?i)^thermonuclear device strike, .*, 4 of 9
key Down -Settle 600
expect (?i)[5-9] of 9
key Down -Settle 600
expect (?i)[5-9] of 9
no-lua-errors
