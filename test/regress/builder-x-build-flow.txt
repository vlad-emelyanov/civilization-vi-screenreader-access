# Merged scripts (one load instead of two), fixture bug-builder-x-list-no-build: both send the builder with X > first tile > Enter and let it build (the tuner restores its moves first),
# the second part again after the first build, with N: after a builder arrives armed, N says what Enter does ("Enter builds <improvement> with the <unit>"), and Enter then does exactly that.
# The second part is blind-written (round 28) and depends on a further tile being offered after the first build; if it fails alone, run it from a fresh load.
# was: builder-x-builds-on-arrival n-matches-armed-build
# covers: **/HubSettle.lua **/HubEvents.lua **/HubAnnounce.lua **/BuildScan.lua **/UnitPanel.lua UI/Additions/InGameAccess.lua
# no-share-load: sends a unit and builds
load bug-builder-x-list-no-build
clear-advisors
# --- part from builder-x-builds-on-arrival ---
lua GameCore_Tuner for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BUILDER" then UnitManager.RestoreMovement(u) end end
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BUILDER" then UI.SelectUnit(u) break end end
key X -Settle 2000
expect ^Tiles to improve near
key Enter -Settle 4000
expect (?i)Arrived.*(Enter builds \w+|can be built next turn: select it and press Enter)
key Enter -Settle 3000
expect (?i)\w+ built, \d charges? left
lua GameCore_Tuner for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BUILDER" and u:GetMovesRemaining()==0 then local i=Map.GetPlot(u:GetX(),u:GetY()):GetImprovementType(); return i>=0 and "improved" or "none" end end
expect-lua ^improved$
lua InGameAccess HubCore.CloseMode( false, true ) return 1
# --- part from n-matches-armed-build ---
lua GameCore_Tuner for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BUILDER" then UnitManager.RestoreMovement(u) end end
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BUILDER" then UI.SelectUnit(u) break end end
key X -Settle 2000
expect ^Tiles to improve near
key Enter -Settle 4000
expect (?i)Arrived.*Enter builds \w+
action KeyNavWhatsNext -Settle 1500
expect (?i)^Enter builds \w+ with the Builder
key Enter -Settle 3000
expect (?i)\w+ built, \d charges? left
# built: nothing armed any more, N is the plain next action again
action KeyNavWhatsNext -Settle 1500
expect-not (?i)^Enter builds
no-lua-errors
