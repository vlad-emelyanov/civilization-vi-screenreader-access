# Verified 2026-10-23. Item 4: RESEARCH COMPLETED is spoken once, not again after an advisor popup.
# The completed popup only comes with the turn rollover: finish the research with the tuner, skip all units (so Enter ends the turn), Enter.
# no-share-load
load early-t60
clear-advisors
lua GameCore_Tuner local t=Players[0]:GetTechs(); for r in GameInfo.Technologies() do if not t:HasTech(r.Index) then t:SetResearchingTech(r.Index); t:ChangeCurrentResearchProgress(5000); break end end
lua InGameAccess local h=GameInfo.UnitOperations["UNITOPERATION_SKIP_TURN"].Hash; for _,u in Players[0]:GetUnits():Members() do UnitManager.RequestOperation(u,h) end
key Enter -Settle 9000
expect (?i)research completed
expect-not (?i)(research|civic) completed.*(research|civic) completed
no-lua-errors
