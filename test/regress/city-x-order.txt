# Recorded round 27. City ranged attack: X cycles attackable enemies (nearest first) before any other unit.
# covers: **/HubCursor.lua **/HubTargets.lua
# no-share-load: scenario spawns units
load bug-ranged-no-reason
clear-advisors
scenario city-x-order
lua InGameAccess for _,c in Players[0]:GetCities():Members() do if Locale.Lookup(c:GetName())=="Ostia" then UI.SelectCity(c) end end
wait Ostia -Timeout 8
action RangedAttack -Settle 1500
expect ^City ranged attack
# the first X may still be the Musketman the cursor starts on; the next one must be the Crossbowman, never the peaceful Builder
key X -Settle 1500
expect (?i)valid target(, .*)?\. Barbarian (Musketman|Crossbowman)
expect-not Builder
key X -Settle 1500
expect (?i)valid target(, .*)?\. Barbarian (Musketman|Crossbowman)
expect-not Builder
# round 27 (M5): X walks only the plots the city can hit, so with two targets the third X wraps to the first
key X -Settle 1500
expect Last target, X wraps|valid target
no-lua-errors
