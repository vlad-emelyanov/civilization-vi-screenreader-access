# Written 2026-10-24. WMD strike with a Bomber: scenario nuke-ready gives one Nuclear Device. The Tab menu lists strike items; the target mode (opened by Enter on the menu item; the items are named by warhead now) lists targets with X; Enter on a target is a real mouse click (#KEYNAV_CLICK) that opens the game's launch confirmation; "Stand down" cancels. NO NUKE IS LAUNCHED (the weapon count stays 1).
# covers: **/HubTargets.lua **/HubCursor.lua **/HubMenus.lua **/UnitPanel.lua
# no-share-load: scenario gives a nuclear device
load air-ready
clear-advisors
# the fixture opens with the Historic moment popup (Taking Flight)
key Enter -Settle 1500
scenario nuke-ready
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_BOMBER" then UI.SelectUnit(u) break end end
key Tab -Settle 1500
expect Bomber actions
key Down -Settle 600
key Down -Settle 600
expect (?i)^Nuclear Device strike, \d of 9
key Enter -Settle 1500
expect (?i)strike: choose a tile
key X -Settle 1500
expect Modern Armor
key Enter -Settle 3000
expect (?i)Commence bombing mission
# Stand down (the default button) cancels; nothing is launched
key Enter -Settle 1500
expect (?i)Launch cancelled
lua GameCore_Tuner return 'warheads='..Players[0]:GetWMDs():GetWeaponCount(0)
expect-lua ^warheads=1$
key Escape -Settle 800
no-lua-errors
