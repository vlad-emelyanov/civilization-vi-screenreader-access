# Gathering Storm). The numpad Divide is refused by `key`, so the mod hotkey KeyNavEraProgress (Control+E) opens the same panel.
# Needs the build with EraProgressPanel_SRA.lua + the new action ("EraProgressPanel: ScreenReaderAccess copy loaded" in Lua.log).
# covers: **/EraProgressPanel_SRA.lua **/keynav_queries_configuration.sql
load medieval-t100
clear-advisors
action KeyNavEraProgress -Settle 2500
expect ^Era progress\. .+, \w+ age\. Era score \d+\. .*Dark age below \d+, .* at \d+\. .*(turns?|Final era)
key Down -Settle 600
expect , 2 of \d+
key Space -Settle 600
expect .{8,}
key End -Settle 600
expect , (\d+) of \1$
key Home -Settle 600
expect ^Era: .+, 1 of \d+
key Escape -Settle 1500
expect ^Era progress closed
action KeyNavEraProgress -Settle 2500
expect ^Era progress\.
action KeyNavEraProgress -Settle 1500
expect ^Era progress closed
# the game's own toggle (numpad Divide = ToggleEraProgress; `key` cannot deliver Divide to the game, so the action handler is called)
lua EraProgressPanel OnInputActionTriggered(Input.GetActionId('ToggleEraProgress'))
expect-lua Era progress. .+ age. Era score
key Escape -Settle 1500
expect ^Era progress closed
no-lua-errors
