# The production panel names the production in progress in its opening line and focuses it; Enter on another item says what it replaces; Tab keeps the focused item by identity.
# covers: **/ProductionPanel.lua **/ProductionText.lua
# no-share-load: sets production
load early-t60
clear-advisors
lua InGame UI.DeselectAllUnits(); UI.SelectCity(Players[0]:GetCities():GetCapitalCity())
key Tab -Settle 2000
expect (?i)choose production
# set a production first (the recommended item), then reopen
key Enter -Settle 2500 -Quiet 700
expect (?i)producing
lua InGame UI.DeselectAllUnits(); UI.SelectCity(Players[0]:GetCities():GetCapitalCity())
key Tab -Settle 2000
expect (?i)choose production.*Currently building
# the game's list hides the production in progress, so focus lands on the first item; End goes to the unavailable-items toggle, Up to the last buildable item
key End -Settle 800
key Up -Settle 800
key Enter -Settle 2500 -Quiet 700
expect (?i)replaces
lua InGame UI.DeselectAllUnits(); UI.SelectCity(Players[0]:GetCities():GetCapitalCity())
key Tab -Settle 2000
key Down -Settle 800
key Tab -Settle 1200
expect (?i)queue
key Escape -Settle 1200
key Escape -Settle 1200
no-lua-errors
