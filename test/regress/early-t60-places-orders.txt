# Places list, Your units: picking a unit that has orders says how to cancel them. Verified 2026-10-23.
# Exploring: "Any order stops exploring, Tab for actions"; fortified: "Shift+C cancels it" (the game offers Cancel, not Wake, for a fortified unit; sleeping ones say "Shift+W wakes it").
# covers: **/HubPlaces.lua
# no-share-load: gives orders
# dirty-group: t60-no-barbarians (the tuner removes the two barbarians that the fixture has next to Rome: they add a Threats category to J; the same idempotent change in every member, so the members run one after the other on one load)
# (also changes visible state: ordered last in its group; the reset check after it is dirty)
load early-t60
clear-advisors
lua GameCore_Tuner for _, u in Players[63]:GetUnits():Members() do local x, y = u:GetX(), u:GetY(); if x == 63 and (y == 33 or y == 34) then Players[63]:GetUnits():Destroy(u) end end return 1
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_SPEARMAN" then UI.SelectUnit(u) break end end
wait Spearman -Timeout 8
action Fortify -Settle 1500
expect ^Fortified
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_SCOUT" and u:GetMovesRemaining()>0 then UI.SelectUnit(u) break end end
wait Scout -Timeout 8
action AutoExplore -Settle 2000
expect ^Exploring automatically
lua InGameAccess UI.DeselectAll()
action KeyNavPlaces -Settle 1500
key Tab -Settle 600
expect ^Your units
key Down -Settle 400 -Count 5
expect Spearman, fortified.*, 1 item, d+ of d+
key Enter -Settle 1500
expect fortified\. Shift\+C cancels it
lua InGameAccess UI.DeselectAll()
action KeyNavPlaces -Settle 1500
key Tab -Settle 600
key Down -Settle 400 -Count 4
expect Scout, exploring
