# covers: **/UnitPanel.lua **/HubTargets.lua

load pt-t128

# the Legion and Settler outside Rome are exposed to the catapults: the end-turn guard (round 24) warns first, the second Enter ends the turn

clear-advisors
lua InGameAccess local p=Players[0]; for _,u in p:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_ROMAN_LEGION" then UI.SelectUnit(u) end end
wait Legion -Timeout 8
key W -Settle 1500
expect ^Attack: .*next target, \S+ attacks, \S+ cancels$

expect ^Cancelled
key W -Settle 1500
expect ^Attack: .*next target, \S+ attacks, \S+ cancels$
key Escape -Settle 1200
expect ^Cancelled

expect ^Attack: .*next target, \S+ attacks, \S+ cancels$

lua InGameAccess UI.DeselectAll()

expect ^No unit selected
