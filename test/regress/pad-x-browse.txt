# Controller X on the bare map with a settler selected and the cursor on it opens the settle browser (the selection hint says "X to browse"). UNVERIFIED, blind-written: run once with civregress --record.
# covers: **/HubGamepad.lua **/HubSettle.lua **/HubCursor.lua
# no-share-load: selects the settler and changes the controller mode
load early-start
clear-advisors
lua InGameAccess for _,u in Players[0]:GetUnits():Members() do if GameInfo.Units[u:GetType()].UnitType=="UNIT_SETTLER" then UI.SelectUnit(u) break end end return 1
# ModeOn (console app: F22)
key F22 -Settle 500
# X (F13): the settler's browser
key F13 -Settle 2000
expect ^Settle spots, \d+ options
key Escape -Settle 800
expect ^Closed
# ModeOff (console app: F23)
key F23 -Settle 500
no-lua-errors
